name: DINKFLIX Build & Release

on:
  workflow_dispatch:
    inputs:
      version:
        description: 'DINKFLIX plugin version (example: 3.0.0.0)'
        required: true
        default: '3.0.0.0'
        type: string
  push:
    tags:
      - 'v*'

permissions:
  contents: write

jobs:
  build-release:
    runs-on: ubuntu-24.04

    steps:
      - name: Checkout repository
        uses: actions/checkout@v7
        with:
          fetch-depth: 0

      - name: Select version
        id: version
        shell: bash
        run: |
          if [ -n "${GITHUB_REF_NAME}" ] && [[ "${GITHUB_REF_TYPE}" == "tag" ]]; then
            version="${GITHUB_REF_NAME#v}"
          else
            version="${{ inputs.version }}"
          fi
          if [[ ! "$version" =~ ^[0-9]+\.[0-9]+\.[0-9]+\.[0-9]+$ ]]; then
            echo "Version must look like 3.0.0.0. Received: $version"
            exit 1
          fi
          echo "version=$version" >> "$GITHUB_OUTPUT"
          echo "tag=v$version" >> "$GITHUB_OUTPUT"

      - name: Verify repository structure
        shell: bash
        run: |
          set -euo pipefail
          required=(
            "src/Jellyfin.Plugin.DinkFlix/Jellyfin.Plugin.DinkFlix.csproj"
            "src/Jellyfin.Plugin.DinkFlix/Plugin.cs"
            "src/Jellyfin.Plugin.DinkFlix/Services/PluginServiceRegistrator.cs"
            "src/Jellyfin.Plugin.DinkFlix/Services/FileTransformationRegistrationService.cs"
            "src/Jellyfin.Plugin.DinkFlix/Services/WebFileTransformation.cs"
            "src/Jellyfin.Plugin.DinkFlix/Web/dinkflix.css"
            "src/Jellyfin.Plugin.DinkFlix/Web/dinkflix.js"
            "src/Jellyfin.Plugin.DinkFlix/Configuration/configPage.html"
            "manifest.json"
          )
          for file in "${required[@]}"; do
            test -f "$file" || { echo "MISSING REQUIRED FILE: $file"; exit 1; }
          done

      - name: Verify frontend copies match
        shell: bash
        run: |
          cmp -s dinkflix.css src/Jellyfin.Plugin.DinkFlix/Web/dinkflix.css || { echo 'Root dinkflix.css does not match embedded frontend CSS.'; exit 1; }
          cmp -s dinkflix.js src/Jellyfin.Plugin.DinkFlix/Web/dinkflix.js || { echo 'Root dinkflix.js does not match embedded frontend JS.'; exit 1; }
          node --check src/Jellyfin.Plugin.DinkFlix/Web/dinkflix.js
          python3 -m json.tool manifest.json >/dev/null

      - name: Setup .NET 10
        uses: actions/setup-dotnet@v6
        with:
          dotnet-version: '10.0.401'

      - name: Restore
        run: dotnet restore src/Jellyfin.Plugin.DinkFlix/Jellyfin.Plugin.DinkFlix.csproj

      - name: Publish plugin
        env:
          PLUGIN_VERSION: ${{ steps.version.outputs.version }}
        run: |
          dotnet publish src/Jellyfin.Plugin.DinkFlix/Jellyfin.Plugin.DinkFlix.csproj \
            --configuration Release \
            --output artifacts/publish \
            --no-restore \
            -p:Version="$PLUGIN_VERSION"

      - name: Verify compiled plugin
        shell: bash
        run: |
          set -euo pipefail
          test -s artifacts/publish/Jellyfin.Plugin.DinkFlix.dll
          echo "Built: $(ls -lh artifacts/publish/Jellyfin.Plugin.DinkFlix.dll)"

      - name: Package plugin
        id: package
        env:
          PLUGIN_VERSION: ${{ steps.version.outputs.version }}
        shell: bash
        run: |
          set -euo pipefail
          mkdir -p artifacts/package
          zip_name="DinkFlix.Web_${PLUGIN_VERSION}.zip"
          rm -f "$zip_name"
          cp artifacts/publish/Jellyfin.Plugin.DinkFlix.dll artifacts/package/
          (cd artifacts/package && zip -9 -j "../../$zip_name" Jellyfin.Plugin.DinkFlix.dll >/dev/null)
          checksum="$(md5sum "$zip_name" | awk '{print toupper($1)}')"
          echo "zip_name=$zip_name" >> "$GITHUB_OUTPUT"
          echo "checksum=$checksum" >> "$GITHUB_OUTPUT"

      - name: Create or update GitHub release
        env:
          GH_TOKEN: ${{ github.token }}
          RELEASE_TAG: ${{ steps.version.outputs.tag }}
          ZIP_NAME: ${{ steps.package.outputs.zip_name }}
          PLUGIN_VERSION: ${{ steps.version.outputs.version }}
        shell: bash
        run: |
          set -euo pipefail
          title="DINKFLIX Web $PLUGIN_VERSION"
          if gh release view "$RELEASE_TAG" >/dev/null 2>&1; then
            gh release upload "$RELEASE_TAG" "$ZIP_NAME" --clobber
            gh release edit "$RELEASE_TAG" --title "$title" --notes "DINKFLIX Web $PLUGIN_VERSION — desktop-first Jellyfin Web experience."
          else
            gh release create "$RELEASE_TAG" "$ZIP_NAME" --target "$GITHUB_SHA" --title "$title" --notes "DINKFLIX Web $PLUGIN_VERSION — desktop-first Jellyfin Web experience."
          fi

      - name: Prepare default branch for manifest update
        shell: bash
        run: |
          set -euo pipefail
          branch="${{ github.event.repository.default_branch }}"
          git fetch origin "$branch"
          git checkout "$branch"
          git pull --ff-only origin "$branch"

      - name: Update repository manifest
        env:
          REPOSITORY: ${{ github.repository }}
          RELEASE_TAG: ${{ steps.version.outputs.tag }}
          PLUGIN_VERSION: ${{ steps.version.outputs.version }}
          CHECKSUM: ${{ steps.package.outputs.checksum }}
        shell: bash
        run: |
          python3 - <<'PY'
          import datetime
          import json
          import os

          path = 'manifest.json'
          with open(path, encoding='utf-8') as fh:
              data = json.load(fh)

          entry = {
              'version': os.environ['PLUGIN_VERSION'],
              'changelog': f"DINKFLIX Web {os.environ['PLUGIN_VERSION']}: custom desktop navigation, Recently Added hero, Continue Watching, Recently Played, Movies, TV Shows, My List, search, title details and lightweight premium styling.",
              'targetAbi': '12.0.0.0',
              'sourceUrl': f"https://github.com/{os.environ['REPOSITORY']}/releases/download/{os.environ['RELEASE_TAG']}/DinkFlix.Web_{os.environ['PLUGIN_VERSION']}.zip",
              'checksum': os.environ['CHECKSUM'],
              'timestamp': datetime.datetime.now(datetime.timezone.utc).replace(microsecond=0).isoformat().replace('+00:00', 'Z')
          }

          found = False
          for plugin in data:
              if plugin.get('guid') == '9b8e4d39-6a4c-4b5d-b6aa-d2b1c4f3c9e8':
                  versions = plugin.setdefault('versions', [])
                  plugin['versions'] = [entry] + [item for item in versions if item.get('version') != entry['version']]
                  found = True
                  break

          if not found:
              raise SystemExit('DINKFLIX plugin entry was not found in manifest.json')

          with open(path, 'w', encoding='utf-8') as fh:
              json.dump(data, fh, indent=2)
              fh.write('\n')
          PY

      - name: Commit manifest back to default branch
        env:
          GH_TOKEN: ${{ github.token }}
          RELEASE_TAG: ${{ steps.version.outputs.tag }}
          PLUGIN_VERSION: ${{ steps.version.outputs.version }}
        shell: bash
        run: |
          set -euo pipefail
          branch="${{ github.event.repository.default_branch }}"
          git config user.name "github-actions[bot]"
          git config user.email "41898282+github-actions[bot]@users.noreply.github.com"
          git add manifest.json
          if git diff --cached --quiet; then
            echo "manifest.json is already current."
          else
            git commit -m "chore: publish DINKFLIX Web $PLUGIN_VERSION"
            git push origin "$branch"
          fi
